dw_direwolf_set_growth_stage_effect = {
    remove_trait = direwolf_pup
    remove_trait = direwolf_young_wolf
    remove_trait = direwolf_adult

    if = {
        limit = { age < 1 }
        add_trait = direwolf_pup
    }
    else_if = {
        limit = { age < 3 }
        add_trait = direwolf_young_wolf
    }
    else = {
        add_trait = direwolf_adult
    }
}

dw_bond_direwolf_effect = {
    set_relation_direwolf_companion = { reason = dw_direwolf_bond_reason  target = $WOLF$ }

    add_trait = direwolf_companion
    add_trait_xp = { trait = direwolf_companion  track = direwolf_companion  value = 10 }

    set_variable = { name = bonded_direwolf  value = $WOLF$ }
    save_temporary_scope_as = dw_new_warg

    $WOLF$ = {
        set_variable = { name = bonded_warg  value = root }

        remove_trait = direwolf_wild
        add_trait = direwolf_bonded
    }
    create_character_memory = {
        type = dw_bonded_direwolf_memory
        participants = { dw_wolf = $WOLF$ }
    }

    dw_direwolf_check_wolf_blooded_dynasty_effect = yes
}

dw_direwolf_mark_dynasty_wolf_blooded_effect = {
    set_variable = dw_wolf_blooded

    every_dynasty_member = {
        limit = { is_alive = yes }
        house ?= {
            if = {
                limit = { NOT = { has_house_modifier = has_direwolf_companion } }
                add_house_modifier = has_direwolf_companion
            }
        }
    }
}

dw_direwolf_check_wolf_blooded_dynasty_effect = {
    if = {
        limit = {
            exists = dynasty
            dynasty ?= { NOT = { has_variable = dw_wolf_blooded } }
            dw_direwolf_dynasty_earns_blood_trigger = yes
        }
        dynasty = { dw_direwolf_mark_dynasty_wolf_blooded_effect = yes }

        if = {
            limit = { is_ai = no }
            send_interface_toast = {
                title = dw_direwolf_dynasty_wolf_blooded_toast
                left_icon = root
            }
        }
    }
}

dw_break_direwolf_bond_effect = {
    if = {
        limit = { has_trait = direwolf }

        if = {
            limit = {
                exists = var:bonded_warg
                var:bonded_warg ?= { has_trait = direwolf_companion }
            }
            var:bonded_warg = { dw_direwolf_grieve_effect = yes }
        }
        else = {
            every_relation = {
                type = direwolf_companion
                limit = { has_trait = direwolf_companion }
                dw_direwolf_grieve_effect = yes
            }
        }

        if = {
            limit = { exists = var:bonded_warg }
            var:bonded_warg ?= {
                remove_trait = direwolf_companion
                remove_variable = bonded_direwolf
            }
        }
        every_relation = {
            type = direwolf_companion
            remove_relation_direwolf_companion = root
            remove_trait = direwolf_companion
            remove_variable = bonded_direwolf
        }
        remove_variable = bonded_warg

        remove_trait = direwolf_bonded
        add_trait = direwolf_wild
    }
    else = {
        if = {
            limit = { exists = var:bonded_direwolf }
            var:bonded_direwolf ?= {
                remove_variable = bonded_warg
                remove_trait = direwolf_bonded
                add_trait = direwolf_wild
            }
        }
        every_relation = {
            type = direwolf_companion
            remove_relation_direwolf_companion = root
            remove_variable = bonded_warg
            remove_trait = direwolf_bonded
            add_trait = direwolf_wild
        }
        remove_trait = direwolf_companion
        remove_variable = bonded_direwolf
    }
}

# Names come from the culture, not from script. Every direwolf template sets
# culture = culture:wolf_culture, so leaving `name` off create_character makes
# the engine draw a gender-correct name from that culture's list -
# common/culture/name_lists/dw_direwolf_name_lists_wolf.txt. Add, remove or
# re-gender names THERE; nothing here needs to change. The canon Stark wolves
# (and their mother Winter) are the only ones still named explicitly, because
# those names are fixed by the books - see dw_direwolf_give_canon_pup_effect.
dw_create_direwolf_effect = {
    create_character = {
        location = $OWNER$.location
        template = $TEMPLATE$
        random_traits = no
        save_scope_as = new_direwolf
    }

    $OWNER$ = {
        add_courtier = scope:new_direwolf
    }

    add_to_global_variable_list = { name = dw_all_wolves  target = scope:new_direwolf }
}

# The two wrappers below are the albino roll, not a name roll - call sites read
# "5 = albino / 95 = ordinary". Both let the culture name the wolf.
dw_direwolf_roll_name_effect = {
    dw_create_direwolf_effect = { OWNER = $OWNER$  TEMPLATE = $TEMPLATE$ }
}

dw_direwolf_roll_albino_name_effect = {
    dw_create_direwolf_effect = { OWNER = $OWNER$  TEMPLATE = $TEMPLATE$ }

}

        age = dw_direwolf_canon_litter_age_value
        mother = scope:dw_canon_mother
        random_traits = no
        save_scope_as = new_direwolf
    }
    add_courtier = scope:new_direwolf
    add_to_global_variable_list = { name = dw_all_wolves  target = scope:new_direwolf }

    scope:new_direwolf = {
        add_trait = direwolf_wild

    }

dw_direwolf_refresh_combat_modifier_effect = {

    dw_direwolf_clear_combat_tiers_effect = yes

    if = {
        limit = {
            dw_has_battle_ready_direwolf_trigger = yes
            has_character_flag = dw_wolf_in_combat
        }

        if = {
            limit = { var:bonded_direwolf = { dw_direwolf_ferocity_value >= 20 } }
            add_character_modifier = dw_bonded_direwolf_combat_modifier_fearsome
        }
        else_if = {
            limit = { var:bonded_direwolf = { dw_direwolf_ferocity_value >= 14 } }
            add_character_modifier = dw_bonded_direwolf_combat_modifier_strong
        }
        else_if = {
            limit = { var:bonded_direwolf = { dw_direwolf_ferocity_value >= 8 } }
            add_character_modifier = dw_bonded_direwolf_combat_modifier
        }
        else = {
            add_character_modifier = dw_bonded_direwolf_combat_modifier_lesser
        }

        if = {
            limit = { NOT = { has_character_modifier = dw_direwolf_guard_modifier } }
            add_character_modifier = dw_direwolf_guard_modifier
        }
    }
    else = {
        remove_character_modifier = dw_direwolf_guard_modifier
    }
}

dw_direwolf_call_to_court_effect = {
    var:bonded_direwolf ?= { save_scope_as = dw_called_wolf }

    if = {
        limit = { exists = scope:dw_called_wolf }

        if = {
            limit = { scope:dw_called_wolf = { dw_direwolf_is_feral_trigger = yes } }
            scope:dw_called_wolf = { dw_direwolf_leave_feral_effect = yes }
        }

        if = {
            limit = {
                dw_pack_ranks_active_trigger = yes
                any_courtier = {
                    has_trait = direwolf
                    dw_direwolf_is_grown_trigger = yes
                    NOT = { this = scope:dw_called_wolf }
                }
            }
            scope:dw_called_wolf = { add_character_flag = dw_forced_outcast }
        }
        else = {
            scope:dw_called_wolf = { remove_character_flag = dw_forced_outcast }
        }

        add_courtier = scope:dw_called_wolf

        scope:dw_called_wolf = {
            remove_trait = direwolf_alpha
            remove_trait = direwolf_beta
            remove_trait = direwolf_omega
            remove_trait = direwolf_outcast
            remove_character_flag = dw_pack_rep
        }

        if = {
            limit = { dw_has_direwolf_pack_trigger = yes }
            dw_direwolf_refresh_pack_full_effect = yes
        }
        else = {
            if = {
                limit = { dw_pack_ranks_active_trigger = yes }
                scope:dw_called_wolf = { add_trait = direwolf_alpha }
            }
            else = {
                scope:dw_called_wolf = { add_character_flag = dw_pack_rep }
            }
            dw_direwolf_publish_pack_effect = yes
        }
    }
}

dw_direwolf_combat_maul_effect = {
    if = {
        limit = {
            exists = var:bonded_direwolf
            var:bonded_direwolf = { is_alive = yes }
            scope:combat_side.enemy_side = {
                any_side_knight = { is_alive = yes }
            }
        }
        var:bonded_direwolf = { save_scope_as = dw_maul_wolf }

        scope:combat_side.enemy_side = {
            random_side_knight = {
                limit = { is_alive = yes }
                save_scope_as = dw_maul_knight
            }
        }

        if = {
            limit = {
                exists = scope:dw_maul_knight
                scope:dw_maul_knight = { is_alive = yes }
            }
            scope:dw_maul_wolf = {
                save_scope_value_as = { name = dw_maul_chance  value = dw_direwolf_maul_win_chance_value }
            }

            random = {
                chance = scope:dw_maul_chance
                set_variable = dw_maul_wolf_won
            }

            if = {
                limit = { has_variable = dw_maul_wolf_won }
                random_list = {
                    45 = {
                        save_scope_value_as = { name = dw_maul_outcome  value = 1 }
                        scope:dw_maul_knight = { add_trait = wounded_2 }
                    }
                    55 = {
                        save_scope_value_as = { name = dw_maul_outcome  value = 2 }
                        scope:dw_maul_knight = {
                            death = {
                                death_reason = death_direwolf_mauled
                                killer = scope:dw_maul_wolf
                            }
                        }
                    }
                }
                remove_variable = dw_maul_wolf_won
            }
            else = {
                random_list = {
                    55 = {
                        save_scope_value_as = { name = dw_maul_outcome  value = 3 }
                        scope:dw_maul_wolf = { add_trait = wounded_2 }
                    }
                    45 = {
                        save_scope_value_as = { name = dw_maul_outcome  value = 4 }
                        scope:dw_maul_wolf = {
                            make_unprunable = yes
                            death = {
                                death_reason = death_battle
                                killer = scope:dw_maul_knight
                            }
                        }
                    }
                }
            }

            add_character_flag = {
                flag = dw_maul_cooldown
                days = 5
            }

            trigger_event = dw_direwolf.1400
        }
    }
}

			add_trait_xp = {
				trait = direwolf_companion
