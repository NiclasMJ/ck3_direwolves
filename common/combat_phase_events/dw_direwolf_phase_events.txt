# Direwolf combat phase events. Root = the bonded character, scope:combat_side
# their side. The wolf is not a combatant itself, so it acts through its master:
# every action has a knight and a commander variant, and the shared 5-day
# dw_maul_cooldown (or dw_wolf_howled / dw_wolf_bolted) keeps both from firing.
# What the wolf does depends on the master's battle stance (var:dw_battle_stance,
# see dw_direwolf_set_battle_stance_effect) and on the wolf's temperament.

dw_direwolf_maul_knight = {
	type = knight

	is_valid = {
		dw_wolf_can_act_in_battle_trigger = yes
		OR = {
			dw_battle_stance_is_hunt_knights_trigger = yes
			dw_battle_stance_is_trigger = { STANCE = guard }
			dw_battle_stance_is_trigger = { STANCE = hunt_commander }
		}
		scope:combat_side.enemy_side = {
			any_side_knight = { is_alive = yes }
		}
	}

	chance = {
		value = dw_direwolf_maul_weight_value
	}

	effect = {
		dw_direwolf_combat_maul_effect = yes
	}
}

	type = knight

	is_valid = {
		dw_wolf_can_act_in_battle_trigger = yes
		dw_battle_stance_is_trigger = { STANCE = hunt_commander }

			exists = side_commander
