# Direwolf combat phase events.
dw_direwolf_maul_knight = {
	type = knight

	is_valid = {
		dw_wolf_can_act_in_battle_trigger = yes
		OR = {
			dw_battle_stance_is_hunt_knights_trigger = yes
			dw_battle_stance_is_trigger = { STANCE = guard }
			dw_battle_stance_is_trigger = { STANCE = hunt_commander }
		}
		scope:combat_side.enemy_side = {
			any_side_knight = { is_alive = yes }
		}
	}

	chance = {
		base = 0
		add = dw_direwolf_maul_weight_value
	}

	effect = {
		dw_direwolf_combat_maul_effect = yes
	}
}
