		add = 2

		add = 4
	}

		limit = { has_game_rule = dw_starting_packs_5 }

	max = 5

dw_direwolf_wolf_blood_value = {
	value = 0

	if = {
		limit = { dw_is_wolf_blooded_trigger = yes }
		add = 100
	}
	else = {
		if = {
			limit = { real_father ?= { dw_is_wolf_blooded_trigger = yes } }
			add = 50
		}
		else = {
			if = {
				limit = { real_father.real_father ?= { dw_is_wolf_blooded_trigger = yes } }
				add = 25
			}
			if = {
				limit = { real_father.real_mother ?= { dw_is_wolf_blooded_trigger = yes } }
				add = 25
			}
		}

		if = {
			limit = { real_mother ?= { dw_is_wolf_blooded_trigger = yes } }
			add = 50
		}
		else = {
			if = {
				limit = { real_mother.real_father ?= { dw_is_wolf_blooded_trigger = yes } }
				add = 25
			}
			if = {
				limit = { real_mother.real_mother ?= { dw_is_wolf_blooded_trigger = yes } }
				add = 25
			}
		}
	}

	min = 0
	max = 100
}
