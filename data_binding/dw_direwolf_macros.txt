macro = {
	description = "Requires a direwolf character"
	definition = "IsCharacterDirewolf"
	replace_with = "Character.HasTrait(GetTrait('direwolf').Self)"
}

macro = {
	description = "Requires a living direwolf character"
	definition = "IsCharacterDirewolfAlive"
	replace_with = "And(IsCharacterDirewolf, Not(Character.IsDeadAndValid))"
}

macro = {
	description = "camera_head, or a direwolf face camera for the wolf's growth stage"
	definition = "DirewolfHeadCamera"
	replace_with = "Select_CString(IsCharacterDirewolf, Select_CString(IsCharacterDirewolfPup, 'camera_direwolf_head_pup', Select_CString(IsCharacterDirewolfYoung, 'camera_direwolf_head_young', 'camera_direwolf_head')), 'camera_head')"
}

}
