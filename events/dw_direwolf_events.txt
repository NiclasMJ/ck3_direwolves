dw_direwolf.0120 = {
	type = character_event
	title = dw_direwolf.0120.t
	desc = dw_direwolf.0120.desc

	theme = family
	override_background = { reference = wilderness_forest }

	left_portrait = {
		character = scope:dw_gift_target
		animation = personality_compassionate
	}

	widget = {
		is_shown = { is_ai = no }
		gui = "agot_character_selection_two_options"
		container = "custom_widgets_container"
	}

	immediate = {
		every_courtier = {
			limit = { dw_direwolf_is_giftable_pup_trigger = yes }
			add_to_list = dw_giftable_pups
		}
		ordered_in_list = {
			list = dw_giftable_pups
			order_by = age
			max = list_size:dw_giftable_pups
			root = {
				add_to_variable_list = {
					name = possible_characters
					target = prev
				}
			}
		}
	}

	option = {
		name = dw_direwolf.0120.a

		if = {
			limit = { exists = var:si_selected_character }
			var:si_selected_character = { save_scope_as = dw_gift_pup }
			dw_direwolf_gift_pup_effect = yes
		}

		if = {
			limit = { has_variable = si_selected_character }
			remove_variable = si_selected_character
		}
		if = {
			limit = { has_variable_list = possible_characters }
			clear_variable_list = possible_characters
		}
	}

	option = {
		name = dw_direwolf.0120.b
		if = {
			limit = { has_variable = si_selected_character }
			remove_variable = si_selected_character
		}
		if = {
			limit = { has_variable_list = possible_characters }
			clear_variable_list = possible_characters
		}
	}
}

	type = character_event
	title = dw_direwolf.0301.t
	desc = dw_direwolf.0301.desc

	theme = hunting
	override_background = { reference = ep2_hunt_snowy_forest }

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:dw_wild_wolf
		animation = direwolf_walk
		camera = camera_direwolf
	}

	immediate = {
		play_music_cue = dw_wolf_howl_single
		random_in_global_list = {
			variable = dw_feral_wolves
			limit = { dw_direwolf_is_northern_feral_trigger = yes }
			weight = {
				base = 100
				modifier = { factor = 0.15  has_character_flag = dw_feral_leader }
			}
			save_scope_as = dw_wild_wolf
		}
	}

	option = {
		name = dw_direwolf.0301.a

		if = {
			limit = { exists = scope:dw_wild_wolf }
			random = {
				chance = dw_direwolf_bond_chance_value

				scope:dw_wild_wolf = { dw_direwolf_leave_feral_effect = yes }
				dw_bond_direwolf_effect = { WOLF = scope:dw_wild_wolf }
				add_courtier = scope:dw_wild_wolf
				dw_direwolf_offer_naming_effect = { WOLF = scope:dw_wild_wolf }
				add_prestige = 200

				random = { chance = 25  add_trait = wounded_1 }
			}

			if = {
				limit = { NOT = { has_trait = direwolf_companion } }
				add_stress = minor_stress_gain
				random = { chance = 35  add_trait = wounded_1 }
			}
		}
		else = {
			add_stress = minor_stress_gain
		}
	}
}

	}
	right_portrait = {

		camera = camera_direwolf
	}

	trigger = {
		is_landed = yes
		any_courtier = {
			has_trait = direwolf
			is_female = yes
			is_alive = yes
			any_relation = { type = direwolf_mate }
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				has_trait = direwolf
				is_female = yes
				is_alive = yes
				any_relation = { type = direwolf_mate }
			}
			save_scope_as = dw_dam
		}
		if = {
			limit = { exists = scope:dw_dam }
			scope:dw_dam = {
				remove_character_modifier = dw_denning_modifier
				random_relation = {
					type = direwolf_mate
					save_scope_as = dw_sire
				}
			}

			if = {
				limit = { exists = scope:dw_sire }
				hidden_effect = {
					dw_direwolf_birth_pup_effect = { SIRE = scope:dw_sire  DAM = scope:dw_dam }
					random = {
						chance = 70
						dw_direwolf_birth_pup_effect = { SIRE = scope:dw_sire  DAM = scope:dw_dam }
					}
					random = {
						chance = 50
						dw_direwolf_birth_pup_effect = { SIRE = scope:dw_sire  DAM = scope:dw_dam }
					}
					random = {
						chance = 25
						modifier = {
							add = 25
							scope:dw_dam = { has_trait = direwolf_massive }
						}
						dw_direwolf_birth_pup_effect = { SIRE = scope:dw_sire  DAM = scope:dw_dam }
					}

					dw_direwolf_refresh_pack_full_effect = yes
				}
			}
		}
	}

	option = {
		name = dw_direwolf.0600.a
		add_prestige = 50
	}

	option = {
		name = dw_direwolf.0600.b
		trigger = {
			any_child = {
				count >= 1
				is_alive = yes
				dw_can_bond_with_direwolf_trigger = yes
				NOT = { has_trait = direwolf_companion }
			}
			any_courtier = {
				has_trait = direwolf_pup
				is_alive = yes
				NOT = { any_relation = { type = direwolf_companion } }
			}
		}
		every_child = {
			limit = {
				is_alive = yes
				dw_can_bond_with_direwolf_trigger = yes
				NOT = { has_trait = direwolf_companion }
			}
			save_scope_as = dw_heir
			root = {
				random_courtier = {
					limit = {
						has_trait = direwolf_pup
						is_alive = yes
						NOT = { any_relation = { type = direwolf_companion } }
					}
					save_scope_as = dw_gift_pup
				}
			}
			if = {
				limit = { exists = scope:dw_gift_pup }
				scope:dw_heir = {
					dw_bond_direwolf_effect = { WOLF = scope:dw_gift_pup }
				}
			}
		}
		add_prestige = 100
	}
}
